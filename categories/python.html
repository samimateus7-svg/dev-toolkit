Web Scraping Boilerplate (BeautifulSoup)

A clean, functional template to fetch HTML content from any website, bypass basic user-agent blocks, and parse elements safely.

import requests
from bs4 import BeautifulSoup

def fetch_data(url):
    headers = {'User-Agent': 'Mozilla/5.0'}
    response = requests.get(url, headers=headers)
    if response.status_code == 200:
        soup = BeautifulSoup(response.text, 'html.parser')
        return soup
    return None
