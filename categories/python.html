Simple Chatbot

A lightweight, keyword-based interactive assistant workflow. Uses dictionary parsing and randomized variations to provide flexible console conversational mechanics.

import random

# Respuestas predefinidas basadas en palabras clave
respuestas_base = {
    "hola": ["¡Hola! ¿En qué puedo ayudarte hoy?", "¡Hola! Qué gusto saludarte."],
    "ayuda": ["Claro, ¿qué necesitas saber?", "Dime, ¿en qué te puedo colaborar?"],
    "precio": ["Nuestros precios varían según el servicio. ¿Podrías darme más detalles?", "Puedes consultar nuestra lista de precios en la página web."],
    "gracias": ["¡De nada! Es un placer ayudarte.", "¡No hay de qué! Que tengas un excelente día."],
    "adios": ["¡Hasta luego!", "¡Adiós! Que estés muy bien, vuelve pronto."]
}

def obtener_respuesta(mensaje):
    mensaje = mensaje.lower()
    
    # Busca palabras clave en el mensaje del usuario
    for palabra, respuestas in respuestas_base.items():
        if palabra in mensaje:
            return random.choice(respuestas)
            
    # Respuesta por defecto si no hay coincidencias
    return "Lo siento, no entiendo esa instrucción. ¿Podrías reformularla?"

def main():
    print("🤖 ¡Hola! Soy tu asistente virtual. Escribe 'salir' para terminar.")
    
    while True:
        # Captura lo que escribe el usuario
        entrada_usuario = input("\nTú: ")
        
        # Condición para salir del chat
        if entrada_usuario.lower() == "salir":
            print("Chatbot: ¡Hasta luego!")
            break
            
        respuesta = obtener_respuesta(entrada_usuario)
        print(f"Chatbot: {respuesta}")

if __name__ == "__main__":
    main()

Web Scraping Boilerplate (BeautifulSoup)

A clean, functional template to fetch HTML content from any website, bypass basic user-agent blocks, and parse elements safely.

import requests
from bs4 import BeautifulSoup

def fetch_data(url):
    headers = {'User-Agent': 'Mozilla/5.0'}
    response = requests.get(url, headers=headers)
    if response.status_code == 200:
        soup = BeautifulSoup(response.text, 'html.parser')
        return soup
    return None

Robust API Request Wrapper with Retries

Executes HTTP requests safely by implementing exponential backoff re-attempts when hitting rate limits or temporary connection drops.

import time
import requests

def safe_request(url, retries=3, backoff=2):
    for i in range(retries):
        try:
            response = requests.get(url, timeout=5)
            if response.status_code == 200:
                return response.json()
        except requests.RequestException:
            if i == retries - 1: raise
            time.sleep(backoff ** i)
    return None

Automated Directory Organizer

Scans a targeted directory path and maps local files into categorized subfolders (Images, Docs, Audio, etc.) based matching file extensions.

import os
import shutil

def organize_folder(target_dir):
    extensions = {
        'Images': ['.jpg', '.jpeg', '.png', '.gif'],
        'Docs': ['.pdf', '.docx', '.txt', '.xlsx'],
        'Audio': ['.mp3', '.wav', '.flac']
    }
    for filename in os.listdir(target_dir):
        filepath = os.path.join(target_dir, filename)
        if os.path.isfile(filepath):
            _, ext = os.path.splitext(filename)
            for folder, ext_list in extensions.items():
                if ext.lower() in ext_list:
                    dest_folder = os.path.join(target_dir, folder)
                    os.makedirs(dest_folder, exist_ok=True)
                    shutil.move(filepath, os.path.join(dest_folder, filename))

    # Simulates operational task workflow

def run_parallel(data_list, workers=4):
    with ThreadPoolExecutor(max_workers=workers) as executor:

System Resource Metrics Logger

Retrieves vital performance telemetry from the underlying operating system environment, exporting simple system allocations metrics cleanly.

import psutil

def get_system_metrics():
    metrics = {
        'cpu_usage_pct': psutil.cpu_percent(interval=1),
        'ram_usage_pct': psutil.virtual_memory().percent,
        'disk_free_gb': psutil.disk_usage('/').free / (1024**3)
    }
    return metrics
