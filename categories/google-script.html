Google Workspace Automation

Auto-Send PDF via Gmail

Converts a specific spreadsheet tab into a PDF and automatically emails it using GmailApp services.

function emailSpreadsheetAsPDF() {
  var sheet = SpreadsheetApp.getActiveSpreadsheet();
  var blob = sheet.getAs('application/pdf').setName("Report.pdf");
  GmailApp.sendEmail("admin@example.com", "Daily Report", "Please find attached.", {
    attachments: [blob]
  });
}

Create Calendar Event

Creates a new event in your primary Google Calendar with specified details directly from data inputs.

function createCalendarEvent() {
  var calendar = CalendarApp.getDefaultCalendar();
  var title = "Project Sync";
  var startTime = new Date("May 25, 2026 10:00:00 EST");
  var endTime = new Date("May 25, 2026 11:00:00 EST");
  calendar.createEvent(title, startTime, endTime, {
    description: "Reviewing dev-toolkit updates."
  });
}

Gmail Bulk Auto-Delete

Automatically filters and clears out old inbox emails while safely preserving key documents using safety word tags.

function borradoMasivoTotal() {
  // =========================================================================
  // 1. CONFIGURACIÓN (Modifica solo esta sección)
  // =========================================================================
  
  // Escribe tus palabras a SALVAR separadas por un espacio
  var palabrasA_Salvar = 'factura contrato banco'; 
  
  // Fecha límite: Borrará TODO lo anterior a este día (Formato: AAAA/MM/DD)
  // Ejemplo: '2025/05/22' (Borrará lo viejo de hace un año hacia atrás)
  var fechaLimite = '2025/05/22'; 
  
  // =========================================================================
  // 2. MOTOR DEL PROGRAMA (Bucle automático de borrado)
  // =========================================================================
  
  // Procesa las palabras clave con el signo menos
  var exclusiones = '';
  if (palabrasA_Salvar.trim() !== '') {
    exclusiones = palabrasA_Salvar.split(' ')
      .map(function(palabra) { return '-"' + palabra.trim() + '"'; })
      .join(' ');
  }
  
  // Construye el filtro combinando la bandeja, la fecha hacia atrás y las exclusiones
  // Ejemplo: "label:inbox before:2025/05/22 -factura"
  var queryFinal = ('label:inbox before:' + fechaLimite + ' ' + exclusiones).trim();
  
  var totalBorrados = 0;
  var continuarBorrando = true;
  
  // Bucle Continuo: Se repite solo hasta que la búsqueda se quede sin correos
  while (continuarBorrando) {
    // Trae lotes de 100 en 100 (El máximo permitido por Google para mover a la papelera)
    var hilos = GmailApp.search(queryFinal, 0, 100);
    
    if (hilos.length > 0) {
      GmailApp.moveThreadsToTrash(hilos);
      totalBorrados += hilos.length;
      
      // Pausa técnica de medio segundo para evitar saturar el servidor de Google
      Utilities.sleep(500); 
    } else {
      // Si ya no encuentra correos en el lote, rompe el ciclo automáticamente
      continuarBorrando = false;
    }
  }
  
  // Registro final en la consola al terminar todo el trabajo
  if (totalBorrados > 0) {
    Logger.log('¡Limpieza Completa! Se borraron automáticamente un total de ' + totalBorrados + ' correos antiguos.');
    Logger.log('Se respetaron los correos que contenían las palabras: ' + palabrasA_Salvar);
  } else {
    Logger.log('Terminado: No se encontraron correos anteriores al ' + fechaLimite + ' para borrar.');
  }
}
