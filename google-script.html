Auto-Send PDF via Gmail

Converts a specific spreadsheet tab into a PDF and automatically emails it using GmailApp services.

function emailSpreadsheetAsPDF() {
  var sheet = SpreadsheetApp.getActiveSpreadsheet();
  var blob = sheet.getAs('application/pdf').setName("Report.pdf");
  GmailApp.sendEmail("admin@example.com", "Daily Report", "Please find attached.", {
    attachments: [blob]
  });
}
